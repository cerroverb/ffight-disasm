# Work RAM variables (0xff0000-0xffffff): ADDRESS NAME. Used for absolute long operands and, in game code, for
# (d16,a5) operands (a5 = 0xffff8000, so ADDRESS = 0xff8000 + d16).
ff12a4 ram_demo_input_ptr       # (-0x6d5c,a5): pointer into the attract demo's recorded input list
ff1514 ram_select_screen       # 1 while the character select screen is up (found by RAM diff, bot/dump_select.lua)
ff1525 ram_select_cursor       # 0 Guy, 2 Cody, 4 Haggar
ff167e ram_exception_code      # set by the exception vector handlers (0xff bus error ... 0x500 trapv)
ff8014 ram_vblank_count        # (0x14,a5): irq2_vblank adds one; loc_002ADA loads it from a table
ff801a ram_sound_queue_write   # (0x1a,a5): index the next queued command goes to; sound_send_queued sends until the read index reaches it
ff801c ram_sound_queue_read    # (0x1c,a5): index of the next command sound_send_queued sends
ff804c ram_credits             # (0x4c,a5): word, the coin routine adds one and stops at 9
ff805c ram_p1_input            # (0x5c,a5): IN1 low byte (player 1 in MAME cps1.cpp cps1_3b), inverted; 0x5d the frame before
ff805e ram_p2_input            # (0x5e,a5): IN1 high byte, inverted; 0x5f the frame before
ff8064 ram_dsw_b               # (0x64,a5): DSWB inverted, copied by read_system_inputs (difficulty, bonus life)
ff8065 ram_dsw_a               # (0x65,a5): DSWA inverted (coinage, 2 coins to start)
ff8067 ram_dsw_c               # (0x67,a5): DSWC inverted (lives bits 0-1, free play bit 2; MAME cps1.cpp ffight ports)
ff807e ram_free_play           # (0x7e,a5): free play DIP, decoded once at start-up (loc_000D46)
ff8085 ram_lives_dip           # (0x85,a5): lives a credit, from the DIP table at dat_000E3E
ff8089 ram_continue_dip        # (0x89,a5): continue allowed, decoded at start-up
ff8184 ram_sound_queue         # (0x184,a5): sound command words, indexed by ram_sound_queue_read and _write
ff8412 ram_screen_left         # (0x412,a5): the camera's left edge (screen_left in beatemup-hitboxes.lua)
ff8568 ram_player1_obj         # 0xc0 byte object slots: players and enemies (mame-rr beatemup-hitboxes.lua)
ff90a8 ram_weapon_obj          # 6 slots of 0xc0: weapons (beatemup-hitboxes.lua, zengfr/romhack CPS1)
ff9528 ram_boss_obj
ffb2e8 ram_container_obj       # 16 slots of 0xc0: containers (same source)
ffbee8 ram_item_obj            # 10 slots of 0xc0: items (same source)
